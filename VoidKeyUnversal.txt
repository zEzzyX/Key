  <script>
    let currentScripts = [];
    let showKeylessOnly = false;

    // Fetch scripts on initial load
    document.addEventListener("DOMContentLoaded", () => {
      fetchLatestScripts();

      // Enable Enter key for search
      document.getElementById("searchInput").addEventListener("keypress", (e) => {
        if (e.key === "Enter") searchScripts();
      });
    });

    // Fetch Latest Scripts from ScriptBlox API
    async function fetchLatestScripts() {
      showLoading();
      try {
        const response = await fetch("https://scriptblox.com/api/script/fetch?page=1");
        const data = await response.json();
        
        if (data.result && data.result.scripts) {
          currentScripts = data.result.scripts;
          document.getElementById("pageSubtitle").innerText = `${data.result.totalScripts || currentScripts.length} scripts available`;
          renderScripts();
        }
      } catch (error) {
        showError("Failed to fetch scripts from ScriptBlox API.");
      }
    }

    // Search Scripts via ScriptBlox API
    async function searchScripts() {
      const query = document.getElementById("searchInput").value.trim();
      if (!query) {
        document.getElementById("pageTitle").innerText = "Latest Scripts";
        fetchLatestScripts();
        return;
      }

      showLoading();
      document.getElementById("pageTitle").innerText = `Search Results`;
      document.getElementById("pageSubtitle").innerText = `Results for "${query}"`;

      try {
        const response = await fetch(`https://scriptblox.com/api/script/search?q=${encodeURIComponent(query)}&page=1`);
        const data = await response.json();
        
        if (data.result && data.result.scripts) {
          currentScripts = data.result.scripts;
          renderScripts();
        } else {
          currentScripts = [];
          renderScripts();
        }
      } catch (error) {
        showError("Error executing search.");
      }
    }

    // Filter Toggle Logic
    function filterType(type) {
      document.getElementById("filterAll").classList.toggle("active", type === 'all');
      document.getElementById("filterKeyless").classList.toggle("active", type === 'keyless');
      showKeylessOnly = (type === 'keyless');
      renderScripts();
    }

    // Render Script Cards
    function renderScripts() {
      const container = document.getElementById("scriptsContainer");
      container.innerHTML = "";

      let filtered = currentScripts;
      if (showKeylessOnly) {
        filtered = currentScripts.filter(s => s.key === false);
      }

      if (filtered.length === 0) {
        container.innerHTML = `<div class="no-results">No scripts found.</div>`;
        return;
      }

      filtered.forEach(script => {
        const card = document.createElement("div");
        card.className = "script-card";

        // Image URL handling (ScriptBlox relative vs absolute)
        let bannerUrl = "https://scriptblox.com/images/default-banner.png";
        if (script.game && script.game.imageUrl) {
          bannerUrl = script.game.imageUrl.startsWith("http") 
            ? script.game.imageUrl 
            : `https://scriptblox.com${script.game.imageUrl}`;
        }

        const gameTitle = script.game ? script.game.name : "Universal";
        const isKeyless = script.key === false;

        card.innerHTML = `
          <img class="card-banner" src="${bannerUrl}" alt="${script.title}" onerror="this.style.display='none'">
          <div class="card-body">
            <div class="card-title">${escapeHtml(script.title)}</div>
            <div class="card-game">Game: ${escapeHtml(gameTitle)}</div>
            <div class="card-tags">
              <span class="tag ${isKeyless ? 'keyless' : ''}">${isKeyless ? 'Keyless' : 'Key Required'}</span>
              ${script.isVerified ? '<span class="tag">Verified</span>' : ''}
              ${script.scriptType ? `<span class="tag">${script.scriptType}</span>` : ''}
            </div>
            <div class="card-actions">
              <button class="btn-copy" onclick="copyScript('${escapeHtml(script.script)}')">Copy Script</button>
            </div>
          </div>
        `;
        container.appendChild(card);
      });
    }

    function copyScript(scriptCode) {
      navigator.clipboard.writeText(scriptCode).then(() => {
        alert("Script copied to clipboard!");
      }).catch(err => {
        alert("Failed to copy script.");
      });
    }

    function showLoading() {
      document.getElementById("scriptsContainer").innerHTML = `<div class="loading">Loading...</div>`;
    }

    function showError(msg) {
      document.getElementById("scriptsContainer").innerHTML = `<div class="no-results">${msg}</div>`;
    }

    function escapeHtml(text) {
      if (!text) return "";
      return text
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#039;");
    }
